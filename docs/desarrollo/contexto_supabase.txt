CREATE TABLE public.bloques_ramos (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  nivel text NOT NULL,
  asignatura text NOT NULL,
  horario text NOT NULL,
  profesor text,
  capacidad_maxima integer DEFAULT 15,
  inscritos_count integer DEFAULT 0,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT bloques_ramos_pkey PRIMARY KEY (id)
);
CREATE TABLE public.profiles (
  id uuid NOT NULL,
  email text NOT NULL,
  role text NOT NULL CHECK (role = ANY (ARRAY['admin'::text, 'docente'::text, 'alumno'::text])),
  full_name text,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
CREATE TABLE public.horarios (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  etiqueta text NOT NULL,
  descripcion text,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT horarios_pkey PRIMARY KEY (id)
);
CREATE TABLE public.matriculas_temporales (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  bloque_horario_id uuid,
  nivel text NOT NULL,
  asignatura text NOT NULL,
  rut_apoderado text,
  nombre_apoderado text,
  email_apoderado text,
  rut_alumno text,
  nombre_alumno text,
  documentos_urls jsonb DEFAULT '[]'::jsonb,
  estado text DEFAULT 'pendiente_pago'::text CHECK (estado = ANY (ARRAY['pendiente_pago'::text, 'pagado'::text, 'expirado'::text, 'cancelado'::text])),
  expires_at timestamp with time zone NOT NULL DEFAULT (now() + '00:10:00'::interval),
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT matriculas_temporales_pkey PRIMARY KEY (id),
  CONSTRAINT matriculas_temporales_bloque_horario_id_fkey FOREIGN KEY (bloque_horario_id) REFERENCES public.horarios(id)
);
