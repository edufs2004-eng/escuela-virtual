CREATE TABLE public.bloques_ramos (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  nivel text NOT NULL,
  asignatura text NOT NULL,
  profesor text,
  capacidad_maxima integer DEFAULT 15,
  inscritos_count integer DEFAULT 0,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  horario_id uuid,
  CONSTRAINT bloques_ramos_pkey PRIMARY KEY (id),
  CONSTRAINT bloques_ramos_horario_id_fkey FOREIGN KEY (horario_id) REFERENCES public.horarios(id)
);
CREATE TABLE public.profiles (
  id uuid NOT NULL,
  email text NOT NULL,
  role text NOT NULL CHECK (role = ANY (ARRAY['admin'::text, 'docente'::text, 'alumno'::text])),
  full_name text,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id)
);
CREATE TABLE public.horarios (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  etiqueta text NOT NULL,
  descripcion text,
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  CONSTRAINT horarios_pkey PRIMARY KEY (id)
);
CREATE TABLE public.matriculas_temporales (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  nivel text NOT NULL,
  asignatura text NOT NULL,
  rut_apoderado text,
  nombre_apoderado text,
  email_apoderado text,
  rut_alumno text,
  nombre_alumno text,
  documentos_urls jsonb DEFAULT '[]'::jsonb,
  estado text DEFAULT 'pendiente_pago'::text CHECK (estado = ANY (ARRAY['pendiente_pago'::text, 'pagado'::text, 'expirado'::text, 'cancelado'::text])),
  expires_at timestamp with time zone NOT NULL DEFAULT (now() + '00:10:00'::interval),
  created_at timestamp with time zone NOT NULL DEFAULT timezone('utc'::text, now()),
  horario_id uuid,
  bloque_ramo_id uuid,
  CONSTRAINT matriculas_temporales_pkey PRIMARY KEY (id),
  CONSTRAINT matriculas_temporales_horario_id_fkey FOREIGN KEY (horario_id) REFERENCES public.horarios(id),
  CONSTRAINT matriculas_temporales_bloque_ramo_id_fkey FOREIGN KEY (bloque_ramo_id) REFERENCES public.bloques_ramos(id)
);
CREATE TABLE public.reservas_matricula (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  clave_idempotencia uuid NOT NULL UNIQUE,
  token_acceso uuid NOT NULL DEFAULT gen_random_uuid() UNIQUE,
  nivel text NOT NULL,
  estado text NOT NULL DEFAULT 'activa'::text CHECK (estado = ANY (ARRAY['activa'::text, 'pago_pendiente'::text, 'confirmada'::text, 'cancelada'::text, 'expirada'::text])),
  expires_at timestamp with time zone NOT NULL DEFAULT (now() + '00:10:00'::interval),
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  pago_iniciado_at timestamp with time zone,
  CONSTRAINT reservas_matricula_pkey PRIMARY KEY (id)
);
CREATE TABLE public.reserva_matricula_bloques (
  reserva_id uuid NOT NULL,
  bloque_ramo_id uuid NOT NULL,
  asignatura text NOT NULL,
  CONSTRAINT reserva_matricula_bloques_pkey PRIMARY KEY (reserva_id, asignatura),
  CONSTRAINT reserva_matricula_bloques_reserva_id_fkey FOREIGN KEY (reserva_id) REFERENCES public.reservas_matricula(id),
  CONSTRAINT reserva_matricula_bloques_bloque_ramo_id_fkey FOREIGN KEY (bloque_ramo_id) REFERENCES public.bloques_ramos(id)
);
